459    /**
460     * Return console input events.
461     */
462    public static Kernel32_INPUT_RECORD[] readConsoleInputHelper(long handle, int count, boolean peek)
463            throws IOException {
464        int[] length = new int[1];
465        int res;
466        long inputRecordPtr = 0;
467        try {
468            inputRecordPtr = malloc(Kernel32_INPUT_RECORD.SIZEOF * count);
469            if (inputRecordPtr == 0) {
470                throw new IOException("cannot allocate memory with JNI");
471            }
472            res = peek
473                    ? PeekConsoleInputW(handle, inputRecordPtr, count, length)
474                    : ReadConsoleInputW(handle, inputRecordPtr, count, length);
475            if (res == 0) {
476                throw new IOException("ReadConsoleInputW failed: " + getLastErrorMessage());
477            }
478            if (length[0] <= 0) {
479                return new Kernel32_INPUT_RECORD[0];
480            }
481            Kernel32_INPUT_RECORD[] records = new Kernel32_INPUT_RECORD[length[0]];
482            for (int i = 0; i < records.length; i++) {
483                records[i] = new Kernel32_INPUT_RECORD();
484                Kernel32_INPUT_RECORD.memmove(
485                        records[i], inputRecordPtr + i * Kernel32_INPUT_RECORD.SIZEOF, Kernel32_INPUT_RECORD.SIZEOF);
486            }
487            return records;
488        } finally {
489            if (inputRecordPtr != 0) {
490                free(inputRecordPtr);
491            }
492        }
493    }

495    /**
496     * Return console input key events (discard other events).
497     *
498     * @param count requested number of events
499     * @return array possibly of size smaller then count
500     */
501    public static Kernel32_INPUT_RECORD[] readConsoleKeyInput(long handle, int count, boolean peek) throws IOException {
502        while (true) {
503            // read events until we have keyboard events, the queue could be full
504            // of mouse events.
505            Kernel32_INPUT_RECORD[] evts = readConsoleInputHelper(handle, count, peek);
506            int keyEvtCount = 0;
507            for (Kernel32_INPUT_RECORD evt : evts) {
508                if (evt.eventType == Kernel32_INPUT_RECORD.KEY_EVENT) keyEvtCount++;
509            }
510            if (keyEvtCount > 0) {
511                Kernel32_INPUT_RECORD[] res = new Kernel32_INPUT_RECORD[keyEvtCount];
512                int i = 0;
513                for (Kernel32_INPUT_RECORD evt : evts) {
514                    if (evt.eventType == Kernel32_INPUT_RECORD.KEY_EVENT) {
515                        res[i++] = evt;
516                    }
517                }
518                return res;
519            }
520        }
521    }

523    public static String getLastErrorMessage() {
524        int errorCode = GetLastError();
525        return getErrorMessage(errorCode);
526    }
527
528    public static String getErrorMessage(int errorCode) {
529        int bufferSize = 160;
530        byte data[] = new byte[bufferSize];
531        FormatMessageW(FORMAT_MESSAGE_FROM_SYSTEM, 0, errorCode, 0, data, bufferSize, null);
532        try {
533            return new String(data, "UTF-16LE").trim();
534        } catch (UnsupportedEncodingException e) {
535            throw new IllegalStateException(e);
536        }
537    }
