221        private static StringBuilder determineStringBuilder(Object args) {
222            if (args == null) {
223                return new StringBuilder(80);
224            } else if (args instanceof Integer) {
225                return new StringBuilder((Integer) args);
226            } else if (args instanceof StringBuilder) {
227                return (StringBuilder) args;
228            } else {
229                throw new IllegalArgumentException("Invalid argument type");
230            }
231        }

581    /**
582     * Moves the cursor to row n, column m. The values are 1-based.
583     * Any values less than 1 are mapped to 1.
584     *
585     * @param row    row (1-based) from top
586     * @param column column (1 based) from left
587     * @return this Ansi instance
588     */
589    public Ansi cursor(final int row, final int column) {
590        return appendEscapeSequence2('H', Math.max(1, row), Math.max(1, column));
591    }
592
593    /**
594     * Moves the cursor to column n. The parameter n is 1-based.
595     * If n is less than 1 it is moved to the first column.
596     *
597     * @param x the index (1-based) of the column to move to
598     * @return this Ansi instance
599     */
600    public Ansi cursorToColumn(final int x) {
601        return appendEscapeSequence1('G', Math.max(1, x));
602    }

604    /**
605     * Moves the cursor up. If the parameter y is negative it moves the cursor down.
606     *
607     * @param y the number of lines to move up
608     * @return this Ansi instance

939    private void flushAttributes() {
940        if (attributeOptions.isEmpty()) return;
941        if (attributeOptions.size() == 1 && attributeOptions.get(0) == 0) {
942            builder.append(FIRST_ESC_CHAR);
943            builder.append(SECOND_ESC_CHAR);
944            builder.append('m');
945        } else {
946            _appendEscapeSequence('m', attributeOptions.toArray());
947        }
948        attributeOptions.clear();
949    }

951    private Ansi _appendEscapeSequence(char command, Object... options) {
952        builder.append(FIRST_ESC_CHAR);
953        builder.append(SECOND_ESC_CHAR);
954        int size = options.length;
955        for (int i = 0; i < size; i++) {
956            if (i != 0) {
957                builder.append(';');
958            }
959            if (options[i] != null) {
960                builder.append(options[i]);
961            }
962        }
963        builder.append(command);
964        return this;
965    }
