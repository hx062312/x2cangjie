023/**
024 * ANSI processor providing <code>process*</code> corresponding to ANSI escape codes.
025 * This class methods implementations are empty: subclasses should actually perform the
026 * ANSI escape behaviors by implementing active code in <code>process*</code> methods.
027 *
028 * <p>For more information about ANSI escape codes, see
029 * <a href="http://en.wikipedia.org/wiki/ANSI_escape_code">Wikipedia article</a>
030 *
031 * @since 1.19
032 */

035    protected final OutputStream os;

041    /**
042     * Helper for processEscapeCommand() to iterate over integer options
043     * @param  optionsIterator  the underlying iterator
044     * @throws IOException      if no more non-null values left
045     */
046    protected int getNextOptionInt(Iterator<Object> optionsIterator) throws IOException {
047        for (; ; ) {
048            if (!optionsIterator.hasNext()) throw new IllegalArgumentException();
049            Object arg = optionsIterator.next();
050            if (arg != null) return (Integer) arg;
051        }
052    }

054    /**
055     * @return true if the escape command was processed.
056     */
057    protected boolean processEscapeCommand(ArrayList<Object> options, int command) throws IOException {
058        try {
059            switch (command) {
060                case 'A':
061                    processCursorUp(optionInt1(options, 0, 1));
062                    return true;
063                case 'B':
064                    processCursorDown(optionInt1(options, 0, 1));
065                    return true;
066                case 'C':
067                    processCursorRight(optionInt1(options, 0, 1));
068                    return true;
069                case 'D':
070                    processCursorLeft(optionInt1(options, 0, 1));
071                    return true;
072                case 'E':
073                    processCursorDownLine(optionInt1(options, 0, 1));
074                    return true;
075                case 'F':
076                    processCursorUpLine(optionInt1(options, 0, 1));
077                    return true;
078                case 'G':
079                    processCursorToColumn(optionInt0(options, 0));
080                    return true;
081                case 'H':
082                case 'f':
083                    processCursorTo(optionInt1(options, 0, 1), optionInt1(options, 1, 1));
084                    return true;
085                case 'J':
086                    processEraseScreen(optionInt1(options, 0, 0));
087                    return true;
088                case 'K':
089                    processEraseLine(optionInt1(options, 0, 0));
090                    return true;
091                case 'L':
092                    processInsertLine(optionInt1(options, 0, 1));
093                    return true;
094                case 'M':
095                    processDeleteLine(optionInt1(options, 0, 1));
096                    return true;
097                case 'S':
098                    processScrollUp(optionInt1(options, 0, 1));
099                    return true;
100                case 'T':
101                    processScrollDown(optionInt1(options, 0, 1));
102                    return true;
103                case 'm':
104                    // Validate all options are ints...
105                    for (Object next : options) {
106                        if (next != null && next.getClass() != Integer.class) {
107                            throw new IllegalArgumentException();
108                        }
109                    }
110
111                    int count = 0;
112                    Iterator<Object> optionsIterator = options.iterator();
113                    while (optionsIterator.hasNext()) {
114                        Object next = optionsIterator.next();
115                        if (next != null) {
116                            count++;
117                            int value = (Integer) next;
118                            if (30 <= value && value <= 37) {
119                                processSetForegroundColor0(value - 30);
120                            } else if (40 <= value && value <= 47) {
121                                processSetBackgroundColor0(value - 40);
122                            } else if (90 <= value && value <= 97) {
123                                processSetForegroundColor1(value - 90, true);
124                            } else if (100 <= value && value <= 107) {
125                                processSetBackgroundColor1(value - 100, true);
126                            } else if (value == 38 || value == 48) {
127                                if (!optionsIterator.hasNext()) {
128                                    continue;
129                                }
130                                // extended color like `esc[38;5;<index>m` or `esc[38;2;<r>;<g>;<b>m`
131                                int arg2or5 = getNextOptionInt(optionsIterator);
132                                if (arg2or5 == 2) {
133                                    // 24 bit color style like `esc[38;2;<r>;<g>;<b>m`
134                                    int r = getNextOptionInt(optionsIterator);
135                                    int g = getNextOptionInt(optionsIterator);
136                                    int b = getNextOptionInt(optionsIterator);
137                                    if (r >= 0 && r <= 255 && g >= 0 && g <= 255 && b >= 0 && b <= 255) {
138                                        if (value == 38) processSetForegroundColorExt1(r, g, b);
139                                        else processSetBackgroundColorExt1(r, g, b);
140                                    } else {
141                                        throw new IllegalArgumentException();
142                                    }
143                                } else if (arg2or5 == 5) {
144                                    // 256 color style like `esc[38;5;<index>m`
145                                    int paletteIndex = getNextOptionInt(optionsIterator);
146                                    if (paletteIndex >= 0 && paletteIndex <= 255) {
147                                        if (value == 38) processSetForegroundColorExt0(paletteIndex);
148                                        else processSetBackgroundColorExt0(paletteIndex);
149                                    } else {
150                                        throw new IllegalArgumentException();
151                                    }
152                                } else {
153                                    throw new IllegalArgumentException();
154                                }
155                            } else {
156                                switch (value) {
157                                    case 39:
158                                        processDefaultTextColor();
159                                        break;
160                                    case 49:
161                                        processDefaultBackgroundColor();
162                                        break;
163                                    case 0:
164                                        processAttributeReset();
165                                        break;
166                                    default:
167                                        processSetAttribute(value);
168                                }
169                            }
170                        }
171                    }
172                    if (count == 0) {
173                        processAttributeReset();
174                    }
175                    return true;
176                case 's':
177                    processSaveCursorPosition();
178                    return true;
179                case 'u':
180                    processRestoreCursorPosition();
181                    return true;
182
183                default:
184                    if ('a' <= command && command <= 'z') {
185                        processUnknownExtension(options, command);
186                        return true;
187                    }
188                    if ('A' <= command && command <= 'Z') {
189                        processUnknownExtension(options, command);
190                        return true;
191                    }
192                    return false;
193            }
194        } catch (IllegalArgumentException ignore) {
195        }
196        return false;
197    }

199    /**
200     * @return true if the operating system command was processed.
201     */
202    protected boolean processOperatingSystemCommand(ArrayList<Object> options) {
203        int command = optionInt0(options, 0);
204        String label = (String) options.get(1);
205        // for command > 2 label could be composed (i.e. contain ';'), but we'll leave
206        // it to processUnknownOperatingSystemCommand implementations to handle that
207        try {
208            switch (command) {
209                case 0:
210                    processChangeIconNameAndWindowTitle(label);
211                    return true;
212                case 1:
213                    processChangeIconName(label);
214                    return true;
215                case 2:
216                    processChangeWindowTitle(label);
217                    return true;
218
219                default:
220                    // not exactly unknown, but not supported through dedicated process methods:
221                    processUnknownOperatingSystemCommand(command, label);
222                    return true;
223            }
224        } catch (IllegalArgumentException ignore) {
225        }
226        return false;
227    }

229    /**
230     * Process character set sequence.
231     * @param options options
232     * @return true if the charcter set select command was processed.
233     */
234    protected boolean processCharsetSelect0(ArrayList<Object> options) {
235        int set = optionInt0(options, 0);
236        char seq = (Character) options.get(1);
237        processCharsetSelect1(set, seq);
238        return true;
239    }
240
241    private int optionInt0(ArrayList<Object> options, int index) {
242        if (options.size() <= index) throw new IllegalArgumentException();
243        Object value = options.get(index);
244        if (value == null) throw new IllegalArgumentException();
245        if (!value.getClass().equals(Integer.class)) throw new IllegalArgumentException();
246        return (Integer) value;
247    }
248
249    private int optionInt1(ArrayList<Object> options, int index, int defaultValue) {
250        if (options.size() > index) {
251            Object value = options.get(index);
252            if (value == null) {
253                return defaultValue;
254            }
255            return (Integer) value;
256        }
257        return defaultValue;
258    }
