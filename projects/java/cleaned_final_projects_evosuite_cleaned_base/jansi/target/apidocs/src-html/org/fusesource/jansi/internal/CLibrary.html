016package org.fusesource.jansi.internal;
017
018/**
019 * Interface to access some low level POSIX functions, loaded by
020 * <a href="http://fusesource.github.io/hawtjni/">HawtJNI</a> Runtime
021 * as <code>jansi</code> library.
022 *
023 * @see JansiLoader
024 */
025@SuppressWarnings("unused")
026public class CLibrary {
027
028    //
029    // Initialization
030    //
031
032    public static final boolean LOADED;
033
034    static {
035        LOADED = JansiLoader.initialize();
036        if (LOADED) {
037            init_();
038        }
039    }
040
041    private static native void init_();
042
043    //
044    // Constants
045    //
046
047    public static int STDOUT_FILENO = 1;
048
049    public static int STDERR_FILENO = 2;
050
051    public static boolean HAVE_ISATTY;
052
053    public static boolean HAVE_TTYNAME;
054
055    public static int TCSANOW;
056    public static int TCSADRAIN;
057    public static int TCSAFLUSH;
058    public static long TIOCGETA;
059    public static long TIOCSETA;
060    public static long TIOCGETD;
061    public static long TIOCSETD;
062    public static long TIOCGWINSZ;
063    public static long TIOCSWINSZ;
064
065    /**
066     * test whether a file descriptor refers to a terminal
067     *
068     * @param fd file descriptor
069     * @return isatty() returns 1 if fd is an open file descriptor referring to a
070     * terminal; otherwise 0 is returned, and errno is set to indicate the
071     * error
072     * @see <a href="http://man7.org/linux/man-pages/man3/isatty.3.html">ISATTY(3) man-page</a>
073     * @see <a href="http://man7.org/linux/man-pages/man3/isatty.3p.html">ISATTY(3P) man-page</a>
074     */
075    public static native int isatty(int fd);
076
077    public static native String ttyname(int filedes);
078
079    /**
080     * The openpty() function finds an available pseudoterminal and returns
081     * file descriptors for the master and slave in amaster and aslave.
082     *
083     * @param amaster master return value
084     * @param aslave  slave return value
085     * @param name    filename return value
086     * @param termios optional pty attributes
087     * @param winsize optional size
088     * @return 0 on success
089     * @see <a href="http://man7.org/linux/man-pages/man3/openpty.3.html">OPENPTY(3) man-page</a>
090     */
091    public static native int openpty(
092            int[] amaster, int[] aslave, byte[] name, CLibrary_Termios termios, CLibrary_WinSize winsize);
093
094    public static native int tcgetattr(int filedes, CLibrary_Termios termios);
095
096    public static native int tcsetattr(int filedes, int optional_actions, CLibrary_Termios termios);
097
098    /**
099     * Control a STREAMS device.
100     *
101     * @see <a href="http://man7.org/linux/man-pages/man3/ioctl.3p.html">IOCTL(3P) man-page</a>
102     */
103    // public static native int ioctl(int filedes, long request, int[] params);
104
105    public static native int ioctl(int filedes, long request, CLibrary_WinSize params);
106
107    /**
108     * Window sizes.
109     *
110     * @see <a href="http://man7.org/linux/man-pages/man4/tty_ioctl.4.html">IOCTL_TTY(2) man-page</a>
111     */
112    public static class CLibrary_WinSize {
113
114        static {
115            JansiLoader.initialize();
116            init_();
117        }
118
119        private static native void init_();
120
121        public static int SIZEOF;
122
123        public short ws_row;
124        public short ws_col;
125        public short ws_xpixel;
126        public short ws_ypixel;
127
128        public CLibrary_WinSize(short ws_row_param, short ws_col_param) {
129            this.ws_row = ws_row_param;
130            this.ws_col = ws_col_param;
131        }
132    }
133
134    /**
135     * termios structure for termios functions, describing a general terminal interface that is
136     * provided to control asynchronous communications ports
137     *
138     * @see <a href="http://man7.org/linux/man-pages/man3/termios.3.html">TERMIOS(3) man-page</a>
139     */
140    public static class CLibrary_Termios {
141
142        static {
143            JansiLoader.initialize();
144            init_();
145        }
146
147        private static native void init_();
148
149        public static int SIZEOF;
150
151        public long c_iflag;
152        public long c_oflag;
153        public long c_cflag;
154        public long c_lflag;
155        public byte[] c_cc = new byte[32];
156        public long c_ispeed;
157        public long c_ospeed;
158    }
159}
