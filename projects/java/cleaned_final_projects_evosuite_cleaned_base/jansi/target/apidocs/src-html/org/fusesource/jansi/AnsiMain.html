038/**
039 * Main class for the library, providing executable jar to diagnose Jansi setup.
040 * <p>If no system property is set and output is sent to a terminal (no redirect to a file):
041 * <ul>
042 * <li>any terminal on any Unix should get <code>RESET_ANSI_AT_CLOSE</code> mode,</li>
043 * <li>on Windows, Git-bash or Cygwin terminals should get <code>RESET_ANSI_AT_CLOSE</code> mode also, since they
044 * support natively ANSI escape sequences like any Unix terminal,</li>
045 * <li>on Windows, cmd.exe, PowerShell or Git-cmd terminals should get <code>WINDOWS</code> mode.</li>
046 * </ul>
047 * If stdout is redirected to a file (<code>&gt; out.txt</code>), System.out should switch to <code>STRIP_ANSI</code>.
048 * Same for stderr redirection (<code>2&gt; err.txt</code>) which should affect System.err mode.
049 * <p>The results will vary if you play with <code>jansi.passthrough</code>, <code>jansi.strip</code> or
050 * <code>jansi.force</code> system property, or if you redirect output to a file.
051 * <p>If you have a specific situation that is not covered, please report precise conditions to reproduce
052 * the issue and ideas on how to detect precisely the affected situation.
053 * @see AnsiConsole
054 */
055public class AnsiMain {
056    public static void main(String... args) throws IOException {
057        System.out.println("Jansi " + getJansiVersion());
058
059        System.out.println();
060
061        // info on native library
062        System.out.println("library.jansi.path= " + System.getProperty("library.jansi.path", ""));
063        System.out.println("library.jansi.version= " + System.getProperty("library.jansi.version", ""));
064        boolean loaded = JansiLoader.initialize();
065        if (loaded) {
066            System.out.println("Jansi native library loaded from " + JansiLoader.getNativeLibraryPath());
067            if (JansiLoader.getNativeLibrarySourceUrl() != null) {
068                System.out.println("   which was auto-extracted from " + JansiLoader.getNativeLibrarySourceUrl());
069            }
070        } else {
071            String prev = System.getProperty(AnsiConsole.JANSI_GRACEFUL);
072            try {
073                System.setProperty(AnsiConsole.JANSI_GRACEFUL, "false");
074                JansiLoader.initialize();
075            } catch (Throwable e) {
076                e.printStackTrace(System.out);
077            } finally {
078                if (prev != null) {
079                    System.setProperty(AnsiConsole.JANSI_GRACEFUL, prev);
080                } else {
081                    System.clearProperty(AnsiConsole.JANSI_GRACEFUL);
082                }
083            }
084        }
085
086        System.out.println();
087
088        System.out.println("os.name= " + System.getProperty("os.name") + ", "
089                + "os.version= " + System.getProperty("os.version") + ", "
090                + "os.arch= " + System.getProperty("os.arch"));
091        System.out.println("file.encoding= " + System.getProperty("file.encoding"));
092        System.out.println("sun.stdout.encoding= " + System.getProperty("sun.stdout.encoding") + ", "
093                + "sun.stderr.encoding= " + System.getProperty("sun.stderr.encoding"));
094        System.out.println("stdout.encoding= " + System.getProperty("stdout.encoding") + ", " + "stderr.encoding= "
095                + System.getProperty("stderr.encoding"));
096        System.out.println("java.version= " + System.getProperty("java.version") + ", "
097                + "java.vendor= " + System.getProperty("java.vendor") + ","
098                + " java.home= " + System.getProperty("java.home"));
099        System.out.println("Console: " + System.console());
100
101        System.out.println();
102
103        System.out.println(AnsiConsole.JANSI_GRACEFUL + "= " + System.getProperty(AnsiConsole.JANSI_GRACEFUL, ""));
104        System.out.println(AnsiConsole.JANSI_MODE + "= " + System.getProperty(AnsiConsole.JANSI_MODE, ""));
105        System.out.println(AnsiConsole.JANSI_OUT_MODE + "= " + System.getProperty(AnsiConsole.JANSI_OUT_MODE, ""));
106        System.out.println(AnsiConsole.JANSI_ERR_MODE + "= " + System.getProperty(AnsiConsole.JANSI_ERR_MODE, ""));
107        System.out.println(AnsiConsole.JANSI_COLORS + "= " + System.getProperty(AnsiConsole.JANSI_COLORS, ""));
108        System.out.println(AnsiConsole.JANSI_OUT_COLORS + "= " + System.getProperty(AnsiConsole.JANSI_OUT_COLORS, ""));
109        System.out.println(AnsiConsole.JANSI_ERR_COLORS + "= " + System.getProperty(AnsiConsole.JANSI_ERR_COLORS, ""));
110        System.out.println(
111                AnsiConsole.JANSI_PASSTHROUGH + "= " + AnsiConsole.getBoolean(AnsiConsole.JANSI_PASSTHROUGH));
112        System.out.println(AnsiConsole.JANSI_STRIP + "= " + AnsiConsole.getBoolean(AnsiConsole.JANSI_STRIP));
113        System.out.println(AnsiConsole.JANSI_FORCE + "= " + AnsiConsole.getBoolean(AnsiConsole.JANSI_FORCE));
114        System.out.println(AnsiConsole.JANSI_NORESET + "= " + AnsiConsole.getBoolean(AnsiConsole.JANSI_NORESET));
115        System.out.println(Ansi.DISABLE + "= " + AnsiConsole.getBoolean(Ansi.DISABLE));
116
117        System.out.println();
118
119        System.out.println("IS_WINDOWS: " + AnsiConsole.IS_WINDOWS);
120        if (AnsiConsole.IS_WINDOWS) {
121            System.out.println("IS_CONEMU: " + AnsiConsole.IS_CONEMU);
122            System.out.println("IS_CYGWIN: " + AnsiConsole.IS_CYGWIN);
123            System.out.println("IS_MSYSTEM: " + AnsiConsole.IS_MSYSTEM);
124        }
125
126        System.out.println();
127
128        diagnoseTty(false); // System.out
129        diagnoseTty(true); // System.err
130
131        AnsiConsole.systemInstall();
132
133        System.out.println();
134
135        System.out.println("Resulting Jansi modes for stout/stderr streams:");
136        System.out.println("  - System.out: " + AnsiConsole.out().toString());
137        System.out.println("  - System.err: " + AnsiConsole.err().toString());
138        System.out.println("Processor types description:");
139        for (AnsiType type : AnsiType.values()) {
140            System.out.println("  - " + type + ": " + type.getDescription());
141        }
142        System.out.println("Colors support description:");
143        for (AnsiColors colors : AnsiColors.values()) {
144            System.out.println("  - " + colors + ": " + colors.getDescription());
145        }
146        System.out.println("Modes description:");
147        for (AnsiMode mode : AnsiMode.values()) {
148            System.out.println("  - " + mode + ": " + mode.getDescription());
149        }
150
151        try {
152            System.out.println();
153
154            testAnsi(false);
155            testAnsi(true);
156
157            if (args.length == 0) {
158                printJansiLogoDemo();
159                return;
160            }
161
162            System.out.println();
163
164            if (args.length == 1) {
165                File f = new File(args[0]);
166                if (f.exists()) {
167                    // write file content
168                    System.out.println(
169                            ansi0().bold().a1("\"" + args[0] + "\" content:").reset());
170                    writeFileContent(f);
171                    return;
172                }
173            }
174
175            // write args without Jansi then with Jansi AnsiConsole
176            System.out.println(ansi0().bold().a1("original args:").reset());
177            int i = 1;
178            for (String arg : args) {
179                AnsiConsole.system_out.print(i++ + ": ");
180                AnsiConsole.system_out.println(arg);
181            }
182
183            System.out.println(ansi0().bold().a1("Jansi filtered args:").reset());
184            i = 1;
185            for (String arg : args) {
186                System.out.print(i++ + ": ");
187                System.out.println(arg);
188            }
189        } finally {
190            AnsiConsole.systemUninstall();
191        }
192    }

199    private static void diagnoseTty(boolean stderr) {
200        int isatty;
201        int width;
202        if (AnsiConsole.IS_WINDOWS) {
203            long console = Kernel32.GetStdHandle(stderr ? Kernel32.STD_ERROR_HANDLE : Kernel32.STD_OUTPUT_HANDLE);
204            int[] mode = new int[1];
205            isatty = Kernel32.GetConsoleMode(console, mode);
206            if ((AnsiConsole.IS_CONEMU || AnsiConsole.IS_CYGWIN || AnsiConsole.IS_MSYSTEM) && isatty == 0) {
207                MingwSupport mingw = new MingwSupport();
208                String name = mingw.getConsoleName(!stderr);
209                if (name != null && !name.isEmpty()) {
210                    isatty = 1;
211                    width = mingw.getTerminalWidth(name);
212                } else {
213                    isatty = 0;
214                    width = 0;
215                }
216            } else {
217                Kernel32.Kernel32_CONSOLE_SCREEN_BUFFER_INFO info = new Kernel32.Kernel32_CONSOLE_SCREEN_BUFFER_INFO();
218                GetConsoleScreenBufferInfo(console, info);
219                width = info.windowWidth();
220            }
221        } else {
222            int fd = stderr ? CLibrary.STDERR_FILENO : CLibrary.STDOUT_FILENO;
223            isatty = CLibrary.LOADED ? CLibrary.isatty(fd) : 0;
224            CLibrary.CLibrary_WinSize ws = new CLibrary.CLibrary_WinSize((short) 0, (short) 0);
225            CLibrary.ioctl(fd, CLibrary.TIOCGWINSZ, ws);
226            width = ws.ws_col;
227        }
228
229        System.out.println("isatty(STD" + (stderr ? "ERR" : "OUT") + "_FILENO): " + isatty + ", System."
230                + (stderr ? "err" : "out") + " " + ((isatty == 0) ? "is *NOT*" : "is") + " a terminal");
231        System.out.println("width(STD" + (stderr ? "ERR" : "OUT") + "_FILENO): " + width);
232    }

234    private static void testAnsi(boolean stderr) {
235        @SuppressWarnings("resource")
236        PrintStream s = stderr ? System.err : System.out;
237        s.print("test on System." + (stderr ? "err" : "out") + ":");
238        for (Ansi.Color c : Ansi.Color.values()) {
239            s.print(" " + ansi0().fg0(c) + c + ansi0().reset());
240        }
241        s.println();
242        s.print("            bright:");
243        for (Ansi.Color c : Ansi.Color.values()) {
244            s.print(" " + ansi0().fgBright(c) + c + ansi0().reset());
245        }
246        s.println();
247        s.print("              bold:");
248        for (Ansi.Color c : Ansi.Color.values()) {
249            s.print(" " + ansi0().bold().fg0(c) + c + ansi0().reset());
250        }
251        s.println();
252        s.print("             faint:");
253        for (Ansi.Color c : Ansi.Color.values()) {
254            s.print(" " + ansi0().a0(Attribute.INTENSITY_FAINT).fg0(c) + c + ansi0().reset());
255        }
256        s.println();
257        s.print("        bold+faint:");
258        for (Ansi.Color c : Ansi.Color.values()) {
259            s.print(" " + ansi0().bold().a0(Attribute.INTENSITY_FAINT).fg0(c) + c + ansi0().reset());
260        }
261        s.println();
262        Ansi ansi = ansi0();
263        ansi.a1("        256 colors: ");
264        for (int i = 0; i < 6 * 6 * 6; i++) {
265            if (i > 0 && i % 36 == 0) {
266                ansi.reset();
267                ansi.newline();
268                ansi.a1("                    ");
269            } else if (i > 0 && i % 6 == 0) {
270                ansi.reset();
271                ansi.a1("  ");
272            }
273            int a0 = i % 6;
274            int a1 = (i / 6) % 6;
275            int a2 = i / 36;
276            ansi.bg1(16 + a0 + a2 * 6 + a1 * 36).a3(' ');
277        }
278        ansi.reset();
279        s.println(ansi);
280        ansi = ansi0();
281        ansi.a1("         truecolor: ");
282        for (int i = 0; i < 256; i++) {
283            if (i > 0 && i % 48 == 0) {
284                ansi.reset();
285                ansi.newline();
286                ansi.a1("                    ");
287            }
288            int r = 255 - i;
289            int g = i * 2 > 255 ? 255 - 2 * i : 2 * i;
290            int b = i;
291            ansi.bgRgb1(r, g, b).fgRgb1(255 - r, 255 - g, 255 - b).a3(i % 2 == 0 ? '/' : '\\');
292        }
293        ansi.reset();
294        s.println(ansi);
295    }

297    private static String getPomPropertiesVersion(String path) throws IOException {
298        InputStream in__ = AnsiMain.class.getResourceAsStream("/META-INF/maven/" + path + "/pom.properties");
299        if (in__ == null) {
300            return null;
301        }
302        try {
303            Properties p = new Properties();
304            p.load(in__);
305            return p.getProperty("version");
306        } finally {
307            closeQuietly(in__);
308        }
309    }
310
311    private static void printJansiLogoDemo() throws IOException {
312        BufferedReader in__ =
313                new BufferedReader(new InputStreamReader(AnsiMain.class.getResourceAsStream("jansi.txt"), UTF_8));
314        try {
315            String l;
316            while ((l = in__.readLine()) != null) {
317                System.out.println(l);
318            }
319        } finally {
320            closeQuietly(in__);
321        }
322    }
323
324    private static void writeFileContent(File f) throws IOException {
325        InputStream in__ = new FileInputStream(f);
326        try {
327            byte[] buf = new byte[1024];
328            int l = 0;
329            while ((l = in__.read(buf)) >= 0) {
330                System.out.write(buf, 0, l);
331            }
332        } finally {
333            closeQuietly(in__);
334        }
335    }
336
337    private static void closeQuietly(Closeable c) {
338        try {
339            c.close();
340        } catch (IOException ioe) {
341            ioe.printStackTrace(System.err);
342        }
343    }
